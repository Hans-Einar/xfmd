cmake_minimum_required(VERSION 3.20)
project(xfmd VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps")
include(CTest)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FOX REQUIRED IMPORTED_TARGET fox>=1.6.57)
find_package(cmark 0.31.1 EXACT CONFIG REQUIRED)
add_library(xfmd_contracts INTERFACE)
target_include_directories(xfmd_contracts INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
option(XFMD_SANITIZERS "Enable Address/UndefinedBehavior sanitizers" OFF)
if(XFMD_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  function(xfmd_test name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE xfmd_contracts)
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
    add_test(NAME ${name} COMMAND ${name})
  endfunction()
  xfmd_test(PortContractTest tests/contracts/PortContractTest.cpp)
  add_test(NAME BlueprintStructure COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_blueprints.py)
endif()
if(BUILD_TESTING)
  add_test(NAME LayerBoundaries COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_layers.py)
endif()
